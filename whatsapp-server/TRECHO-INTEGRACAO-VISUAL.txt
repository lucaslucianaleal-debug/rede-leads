async function sendReminderToWhatsApp(phoneId, reminderText) {
  try {
    // Fazer POST para http://localhost:3000/send-message
    const response = await fetch(`${BACKEND_URL}/send-message`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        phone: phoneId,              // ← ID canônico (11 dígitos)
        message: reminderText,       // ← generateReminderText(dataAgendamento, type)
        isReminder: true             // ← flag para diferenciar de mensagens manuais
      })
    });

    // Verificar se o request foi bem-sucedido
    if (!response.ok) {
      const errorText = await response.text();
      throw new Error(`HTTP ${response.status}: ${errorText}`);
    }

    // Parse response
    const data = await response.json();
    console.log(`[reminder-worker] ✅ Lembrete enviado com sucesso para ${phoneId}`);
    
    return true;  // 🎯 Sucesso → será marcado em Firestore
    
  } catch (error) {
    // Qualquer erro (conexão, timeout, HTTP error, etc)
    console.error(`[reminder-worker] ❌ Erro ao enviar lembrete para ${phoneId}:`, error.message);
    
    return false;  // 🎯 Falha → NÃO será marcado (retry próxima rodada)
  }
}
